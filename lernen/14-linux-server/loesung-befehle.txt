# Einheit 14 – Musterlösung Befehle (andere Schreibweisen sind oft auch richtig)
# Mit oder ohne sudo, wo es egal ist. Leer lassen = noch offen.
# Prüfen: node --test lernen/14-linux-server/

# Als Benutzer pi auf dem Pi mit der IP 192.168.100.133 anmelden
ssh-anmelden: ssh pi@192.168.100.133

# Nachsehen, ob der Dienst nodered läuft
status-nodered: systemctl status nodered

# Den Dienst grafana-server neu starten
neustart-grafana: sudo systemctl restart grafana-server

# Dafür sorgen, dass mariadb beim Hochfahren automatisch startet
autostart-mariadb: sudo systemctl enable mariadb

# Das Log von nodered live mitlesen
log-live-nodered: journalctl -u nodered -f

# Die Paketlisten aktualisieren
paketlisten: sudo apt update

# Das Paket libatomic1 installieren
installieren-libatomic: sudo apt install libatomic1

# Die Datei planeten-erdbeben-setup-linux-arm64 ausführbar machen
ausfuehrbar: chmod +x planeten-erdbeben-setup-linux-arm64

# Die Datei .env so einstellen, dass nur der Besitzer sie lesen und schreiben darf (als Zahl)
nur-besitzer-env: chmod 600 .env

# Den freien Speicherplatz in lesbarer Form (GB/MB) anzeigen
speicherplatz: df -h

# Anzeigen, welche Programme auf welchen TCP-Ports lauschen (mit ss)
lauschende-ports: sudo ss -ltnp

# Die Datenbank planeten_erdbeben als Benutzer planeten in die Datei backup.sql sichern
datenbank-sichern: mysqldump -u planeten -p planeten_erdbeben > backup.sql

# cron-Zeile: das Skript /home/pi/backup.sh jeden Tag um 02:30 ausführen
cron-zeile: 30 2 * * * /home/pi/backup.sh
